cmake_minimum_required(VERSION 3.25)
project(experiment_instrument VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
FetchContent_Declare(asio GIT_REPOSITORY https://github.com/chriskohlhoff/asio.git
  GIT_TAG 12b52a54a25d1861b037bc5d9810f92a7151d920)
FetchContent_Declare(doctest GIT_REPOSITORY https://github.com/doctest/doctest.git
  GIT_TAG 2d0a9359a60c51affe2a9bebb1be1dca47868151)
FetchContent_MakeAvailable(json asio doctest)
find_package(Threads REQUIRED)
add_library(lab_core src/engine.cpp)
target_include_directories(lab_core PUBLIC src)
target_link_libraries(lab_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(lab_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(instrument src/server.cpp)
target_include_directories(instrument SYSTEM PRIVATE ${asio_SOURCE_DIR}/asio/include)
target_compile_definitions(instrument PRIVATE ASIO_STANDALONE)
target_link_libraries(instrument PRIVATE lab_core Threads::Threads)
target_link_options(instrument PRIVATE -static-libstdc++ -static-libgcc)
target_compile_options(instrument PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(LAB_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
if(LAB_SANITIZERS)
  target_compile_options(lab_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lab_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(engine_tests tests/engine_test.cpp)
target_link_libraries(engine_tests PRIVATE lab_core doctest::doctest)
add_test(NAME engine_tests COMMAND engine_tests)
